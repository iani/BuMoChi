// Forward bunraku data to osc addresses
// In response to changed \msg messages from
// OscSequence listening to OscController.

BunrakuForwarder : NamedInstance {
	var <>addr; // addresses to forward to
	var filters; // array of ValueAdapters for
	var <>msgCache;
	var <isEnabled = false;
	// filtering (modifying) the bunraku osc message array.

	enable {
		addr.forwardMsg('/bunraku/avatar/frame', filters, this);
		isEnabled = true;
	}

	disable {
		addr unforwardMsg: '/bunraku/avatar/frame';
		isEnabled = false;
	}

	addFilter { | filterFunc, jointName, offset = 1 |
		var index;
		index = msgCache indexOf: jointName;
		this.filters[index + offset] = filterFunc.asFilter;
		this.filters = filters;
	}

	filters {
		^filters ?? { filters = nil ! msgCache.size };
	}

	filters_ { | argFilters |
		filters = argFilters;
		if (isEnabled) { // update filters in forward action
			this.disable;
			this.enable;
		}
	}
}
